48 float parabolicInterpolation(int tauEstimate) {
49 if (tauEstimate < 1 || tauEstimate >= halfBufferSize - 1) return tauEstimate;
50 float s0 = yinBuffer[tauEstimate - 1];
51 float s1 = yinBuffer[tauEstimate];
52 float s2 = yinBuffer[tauEstimate + 1];
53 return tauEstimate + (s2 - s0) / (2.0f * (2.0f * s1 - s2 - s0));
54 }

61 float getMedianPitch(float newPitch) {
62 if (newPitch > 0) {
63 silenceCount = 0;
64 pitchHistory.push_back(newPitch);
65 if (pitchHistory.size() > historySize) {
66 pitchHistory.pop_front();
67 }
68 } else {
69 silenceCount++;
70 if (silenceCount > maxSilenceFrames) {
71 pitchHistory.clear();
72 }
73 return -1.0f;
74 }
75
76 if (pitchHistory.empty()) return -1.0f;
77
78 std::vector<float> sortedHistory(pitchHistory.begin(), pitchHistory.end());
79 std::sort(sortedHistory.begin(), sortedHistory.end());
80
81 return sortedHistory[sortedHistory.size() / 2];
82 }

100 float detectPitch(const float* audioBuffer) {
101 float sumSquares = 0.0f;
102 for (int i = 0; i < bufferSize; i++) {
103 sumSquares += audioBuffer[i] * audioBuffer[i];
104 }
105 float rms = std::sqrt(sumSquares / bufferSize);
106
107 if (rms < 0.001f) {
108 return getMedianPitch(-1.0f);
109 }
110
111 for (int tau = 0; tau < halfBufferSize; tau++) {
112 yinBuffer[tau] = 0.0f;
113 for (int i = 0; i < halfBufferSize; i++) {
114 float delta = audioBuffer[i] - audioBuffer[i + tau];
115 yinBuffer[tau] += delta * delta;
116 }
117 }
118
119 yinBuffer[0] = 1.0f;
120 float runningSum = 0.0f;
121 for (int tau = 1; tau < halfBufferSize; tau++) {
122 runningSum += yinBuffer[tau];
123 yinBuffer[tau] = yinBuffer[tau] * tau / runningSum;
124 }
125
126 int tauEstimate = -1;
127 for (int tau = 1; tau < halfBufferSize; tau++) {
128 if (yinBuffer[tau] < threshold) {
129 while (tau + 1 < halfBufferSize && yinBuffer[tau + 1] < yinBuffer[tau]) {
130 tau++;
131 }
132 tauEstimate = tau;
133 break;
134 }
135 }
136
137 if (tauEstimate != -1) {
138 float betterTau = parabolicInterpolation(tauEstimate);
139 return getMedianPitch(sampleRate / betterTau);
140 }
141
142 return getMedianPitch(-1.0f);
143 }
