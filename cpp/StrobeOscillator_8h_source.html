45 StrobeOscillator(float sampleRate, float cutoffHz = 5.0f)
46 : sampleRate(sampleRate), currentPhase(0.0f), targetFrequency(0.0f),
47 filteredI(0.0f), filteredQ(0.0f) {
48
49 float dt = 1.0f / sampleRate;
50 float rc = 1.0f / (2.0f * M_PI * cutoffHz);
51 alpha = dt / (rc + dt);
52 }

58 void setTargetFrequency(float freq) {

74 void processBuffer(const float* audioBuffer, int bufferSize, float& outI, float& outQ) {
75 if (targetFrequency <= 0.0f) {
76 outI = 0.0f;
77 outQ = 0.0f;
78 return;
79 }
80
81 float phaseIncrement = 2.0f * M_PI * targetFrequency / sampleRate;
82
83 for (int i = 0; i < bufferSize; ++i) {
84 float sample = audioBuffer[i];
85
86 float loI = std::cos(currentPhase);
87 float loQ = std::sin(currentPhase);
88
89 filteredI = alpha * (sample * loI) + (1.0f - alpha) * filteredI;
90 filteredQ = alpha * (sample * loQ) + (1.0f - alpha) * filteredQ;
91
92 currentPhase += phaseIncrement;
93 if (currentPhase >= 2.0f * M_PI) {
94 currentPhase -= 2.0f * M_PI;
95 }
96 }
97
98 outI = filteredI;
99 outQ = filteredQ;
100 }
